function goalScore(g: OrganismGoal): number {
  return clamp(
    g.importance * 0.34 +
    g.urgency * 0.18 +
    g.confidence * 0.16 +
    (1 - g.progress) * 0.12 +
    (g.source === 'intrinsic' || g.source === 'homeostatic' ? 0.12 : 0.05)
  );
}

function needRelief(action: ActionKind, n: OrganismNeeds): number {
  const relief = {
    respond: n.socialNeed * 0.7 + n.goalPressure * 0.2,
    explore: n.novelty * 0.65 + n.uncertainty * 0.55,
    verify: n.uncertainty * 0.85 + n.identityCoherence * 0.08,
    rest: (1 - n.energy) * 0.95 + n.cognitiveLoad * 0.55,
    reflect: n.identityCoherence * 0.35 + n.uncertainty * 0.35 + n.goalPressure * 0.18,
    wait: 0.2 + (1 - n.cognitiveLoad) * 0.2,
    defer: n.cognitiveLoad * 0.3 + n.uncertainty * 0.25,
  }[action];
  return clamp(relief);
}

function actionCost(action: ActionKind): number {
  return {
    respond: 0.18,
    explore: 0.42,
    verify: 0.36,
    rest: 0.05,
    reflect: 0.22,
    wait: 0.02,
    defer: 0.08,
  }[action];
}

function predictedUtility(action: ActionKind, g: OrganismGoal | null, n: OrganismNeeds): number {
  const goalUtility = g ? goalScore(g) : 0;
  const relief = needRelief(action, n);
  const cost = actionCost(action);
  const autonomyBonus = action === 'explore' || action === 'verify' || action === 'reflect' ? 0.06 : 0;
  return clamp(goalUtility * 0.54 + relief * 0.46 + autonomyBonus - cost * 0.26);
}
